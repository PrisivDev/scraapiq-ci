   182→   182→            description: data.progress.errors[0] || "Erreur inconnue",
   183→   183→          })
   184→   184→        }
   185→   185→      }
   186→   186→    } catch (err) {
   187→   187→      console.error("[poll] error:", err)

  1045→  1045→function formatEvent(event: { type: string; [key: string]: unknown }): string {

  1047→  1047→    // Google Maps events
  1048→  1048→    case "start":
  1049→  1049→      return "🚀 Démarrage du scraping"
  1050→  1050→    case "search-loaded":
  1051→  1051→      return `🔍 Page de recherche chargée (${event.resultsOnPage} résultats visibles)`
  1052→  1052→    case "scroll":

  1054→  1054→    case "place-extracted":

  1056→  1056→    case "duplicate-detected":
  1057→  1057→      return `↺ Doublon détecté: ${(event.place as { name?: string })?.name || ""} → ${event.duplicateOf}`

  1061→  1061→      return `✗ Erreur: ${event.message}`
  1062→  1062→    case "progress":
  1063→  1063→      return `📊 Phase: ${event.phase} (${event.progress}%)`
  1064→  1064→    case "complete":
  1065→  1065→      return `✅ Terminé : ${event.results ? (event.results as unknown[]).length : 0} lieux, ${event.duplicates} doublons, en ${(Number(event.durationMs) / 1000).toFixed(1)}s`
  1066→  1066→    case "cancelled":
  1067→  1067→      return "⏹️ Annulé"
  1068→  1068→    // Facebook events
  1069→  1069→    case "fb-login-required":
  1070→  1070→      return "🔐 Connexion Facebook requise"
  1071→  1071→    case "fb-consent-required":
