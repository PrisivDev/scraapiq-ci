   249→function formatDate(date: string | Date | null | undefined, pattern = "d MMM yyyy"): string {
   250→  if (!date) return "—"
   251→  try {
   252→    const d = typeof date === "string" ? new Date(date) : date
   253→    if (isNaN(d.getTime())) return "—"
   254→    return format(d, pattern, { locale: fr })
   255→  } catch {
   256→    return "—"
   257→  }
   258→}
   259→
   260→function formatDateTime(date: string | Date | null | undefined): string {
   261→  return formatDate(date, "d MMM yyyy 'à' HH:mm")
   262→}
   263→
   264→function formatCurrency(amount: number | null | undefined): string {
   265→  if (amount === null || amount === undefined) return "—"
   266→  return new Intl.NumberFormat("fr-FR").format(amount) + " FCFA"
   267→}
   268→
   269→function formatNumber(n: number | null | undefined): string {
   270→  if (n === null || n === undefined) return "0"
   271→  return new Intl.NumberFormat("fr-FR").format(n)
   272→}
   273→
   274→function getInitials(name: string | null | undefined, email?: string | null): string {
   275→  const src = name || email || "?"
   276→  const parts = src.split(/[\s@._-]+/).filter(Boolean)
   277→  if (parts.length >= 2) return (parts[0][0] + parts[1][0]).toUpperCase()
   278→  return src.slice(0, 2).toUpperCase()
   279→}

   300→function statusBadgeClass(status: string): string {
   301→  const s = status.toLowerCase()
   302→  if (["active", "paid", "success", "delivered", "ready"].includes(s)) {
   303→    return "bg-emerald-100 text-emerald-700 dark:bg-emerald-900/30 dark:text-emerald-400"
   304→  }
   305→  if (["pending", "trialing", "warning", "warn", "past_due"].includes(s)) {
   306→    return "bg-amber-100 text-amber-700 dark:bg-amber-900/30 dark:text-amber-400"
   307→  }
   308→  if (["revoked", "cancelled", "expired", "failed", "void", "suspended", "error"].includes(s)) {
   309→    return "bg-red-100 text-red-700 dark:bg-red-900/30 dark:text-red-400"
   310→  }
   311→  return "bg-slate-200 text-slate-700 dark:bg-slate-700 dark:text-slate-200"
   312→}
   313→
   314→function severityBadgeClass(severity: string): string {
   315→  const s = severity.toLowerCase()
   316→  if (s === "critical" || s === "error") return "bg-red-100 text-red-700 dark:bg-red-900/30 dark:text-red-400"
   317→  if (s === "warn") return "bg-amber-100 text-amber-700 dark:bg-amber-900/30 dark:text-amber-400"
   318→  if (s === "info") return "bg-sky-100 text-sky-700 dark:bg-sky-900/30 dark:text-sky-400"
   319→  return "bg-slate-200 text-slate-700 dark:bg-slate-700 dark:text-slate-200"
   320→}
   321→
   322→function progressColorClass(pct: number): string {
   323→  if (pct >= 90) return "bg-red-500"
   324→  if (pct >= 60) return "bg-amber-500"
   325→  return "bg-emerald-500"
   326→}

   435→export function OrganizationDetailsDialog({ open, onOpenChange }: OrgDetailsDialogProps) {
   436→  const [loading, setLoading] = React.useState(false)
   437→  const [error, setError] = React.useState<string | null>(null)
   438→  const [data, setData] = React.useState<OrgDetailsResponse | null>(null)
   439→  const [activeTab, setActiveTab] = React.useState("identity")
   440→
   441→  // Inline edit state (org name)
   442→  const [editingName, setEditingName] = React.useState(false)
   443→  const [nameValue, setNameValue] = React.useState("")
   444→  const [savingName, setSavingName] = React.useState(false)
   445→
   446→  const fetchData = React.useCallback(async () => {
   447→    setLoading(true)
   448→    setError(null)
   449→    try {
   450→      const res = await fetch("/api/organization/details", { credentials: "include" })
   451→      if (!res.ok) {
   452→        const j = await res.json().catch(() => null)
   453→        throw new Error(j?.error || `HTTP ${res.status}`)
   454→      }
   455→      const json: OrgDetailsResponse = await res.json()
   456→      setData(json)
   457→      setNameValue(json.organization.name)
   458→    } catch (e) {
   459→      setError(e instanceof Error ? e.message : "Erreur inconnue")
   460→    } finally {
   461→      setLoading(false)
   462→    }
   463→  }, [])
   464→
   465→  React.useEffect(() => {
   466→    if (open) {
   467→      void fetchData()
   468→    }
   469→  }, [open, fetchData])
   470→
   471→  const handleSaveName = async () => {
   472→    if (!data) return
   473→    const trimmed = nameValue.trim()
   474→    if (!trimmed || trimmed.length > 100) {
   475→      toast.error("Nom invalide (1 à 100 caractères)")
   476→      return
   477→    }
   478→    if (trimmed === data.organization.name) {
   479→      setEditingName(false)
   480→      return
   481→    }
   482→    setSavingName(true)
   483→    try {
   484→      const res = await fetch("/api/organization", {
   485→        method: "PUT",
   486→        headers: { "Content-Type": "application/json" },
   487→        credentials: "include",
   488→        body: JSON.stringify({ name: trimmed }),
   489→      })
   490→      if (!res.ok) {
   491→        const j = await res.json().catch(() => null)
   492→        throw new Error(j?.error || `HTTP ${res.status}`)
   493→      }
   494→      toast.success("Organisation renommée", { description: `« ${trimmed} »` })
   495→      setEditingName(false)
   496→      await fetchData()
   497→    } catch (e) {
   498→      toast.error("Erreur", { description: e instanceof Error ? e.message : "Inconnue" })
   499→    } finally {
   500→      setSavingName(false)
   501→    }
   502→  }
   503→
   504→  const canEdit = data?.currentRole === "OWNER" || data?.currentRole === "ADMIN"
   505→
   506→  return (
   507→    <Dialog open={open} onOpenChange={onOpenChange}>
   508→      <DialogContent className="sm:max-w-4xl p-0 gap-0 max-h-[90vh] flex flex-col overflow-hidden">
   509→        <DialogHeader className="px-6 pt-6 pb-4 border-b">
   510→          <div className="flex items-start justify-between gap-3 pr-8">
   511→            <div className="flex items-start gap-3 min-w-0">
   512→              <div className="flex h-12 w-12 items-center justify-center rounded-lg bg-primary text-primary-foreground shrink-0">
   513→                <Building2 className="h-6 w-6" />
   514→              </div>
   515→              <div className="min-w-0">
   516→                <DialogTitle className="text-xl truncate">
   517→                  {loading ? "Chargement…" : data?.organization.name || "Organisation"}
   518→                </DialogTitle>
   519→                <DialogDescription className="text-xs flex flex-wrap items-center gap-x-2 gap-y-0.5">
   520→                  {data && (
   521→                    <>
   522→                      <Badge variant="outline" className={cn("text-[10px] capitalize", PLAN_BADGE[data.organization.plan] || statusBadgeClass(data.organization.plan))}>
   523→                        {data.organization.plan}
   524→                      </Badge>
   525→                      <span className="inline-flex items-center gap-1">
   526→                        <Hash className="h-3 w-3" />
   527→                        {data.organization.slug}
   528→                      </span>
   529→                      <span className="inline-flex items-center gap-1">
   530→                        <Calendar className="h-3 w-3" />
   531→                        Membre depuis {formatDate(data.stats.memberSince)} ({data.stats.daysActive} jour{data.stats.daysActive > 1 ? "s" : ""})
   532→                      </span>
   533→                    </>
   534→                  )}
   535→                </DialogDescription>
   536→              </div>
   537→            </div>
   538→          </div>
   539→        </DialogHeader>
   540→
   541→        <div className="flex-1 overflow-y-auto px-6 py-4 space-y-4">
   542→          {loading && <LoadingState />}
   543→          {error && !loading && (
   544→            <div className="flex flex-col items-center justify-center text-center py-16">
   545→              <div className="flex h-12 w-12 items-center justify-center rounded-2xl bg-red-100 text-red-600 dark:bg-red-900/30 dark:text-red-400 mb-3">
   546→                <AlertCircle className="h-6 w-6" />
   547→              </div>
   548→              <h4 className="text-base font-semibold mb-1">Impossible de charger les détails</h4>
   549→              <p className="text-sm text-muted-foreground mb-3">{error}</p>
   550→              <Button onClick={() => void fetchData()} variant="outline" size="sm">
   551→                <Loader2 className="h-3.5 w-3.5 mr-2" />
   552→                Réessayer
   553→              </Button>
   554→            </div>
   555→          )}
   556→
   557→          {data && !loading && !error && (
   558→            <>
   559→              {/* KPI grid */}
   560→              <div className="grid grid-cols-2 sm:grid-cols-3 gap-2">
   561→                <KpiCard
   562→                  label="Membres actifs"
   563→                  value={formatNumber(data.stats.totalMembers)}
   564→                  sub={`${data.stats.totalMembersPending} en attente`}
   565→                  icon={Users}
   566→                />
   567→                <KpiCard
   568→                  label="Workspaces"
   569→                  value={formatNumber(data.stats.totalWorkspaces)}
   570→                  icon={Boxes}
   571→                />
   572→                <KpiCard
   573→                  label="Plan"
   574→                  value={data.organization.plan}
   575→                  sub={data.license ? "Licence active" : "Sans licence"}
   576→                  icon={ShieldCheck}
   577→                />
   578→                <KpiCard
   579→                  label="Quota API"
   580→                  value={data.quota ? `${formatNumber(data.quota.apiCalls)} / ${formatNumber(data.license?.maxApiCalls ?? 100000)}` : "0"}
   581→                  sub={data.quota ? `${data.quota.periodMonth}/${data.quota.periodYear}` : "Aucun quota"}
   582→                  icon={Gauge}
   583→                />
   584→                <KpiCard
   585→                  label="Entreprises"
   586→                  value={formatNumber(data.stats.totalCompanies)}
   587→                  sub={data.license ? `/ ${formatNumber(data.license.maxCompanies)}` : "global"}
   588→                  icon={Layers}
   589→                />
   590→                <KpiCard
   591→                  label="Jours actifs"
   592→                  value={formatNumber(data.stats.daysActive)}
   593→                  icon={Activity}
   594→                />
   595→              </div>
   596→
   597→              {/* Tabs */}
   598→              <Tabs value={activeTab} onValueChange={setActiveTab} className="w-full">
   599→                <div className="overflow-x-auto -mx-1 px-1">
   600→                  <TabsList className="h-auto flex-wrap">
   601→                    <TabsTrigger value="identity" className="gap-1">
   602→                      <Building2 className="h-3.5 w-3.5" /> Identité
   603→                    </TabsTrigger>
