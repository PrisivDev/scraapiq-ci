   883→   883→function Sparkline({ data, color }: { data: number[]; color: string }) {
   884→   884→  const max = Math.max(...data)
   885→   885→  const min = Math.min(...data)
   886→   886→  const range = max - min || 1
   887→   887→  const w = 60
   888→   888→  const h = 20
   889→   889→  const points = data.map((v, i) => `${(i / (data.length - 1)) * w},${h - ((v - min) / range) * h}`).join(" ")
   890→   890→  return (
   891→   891→    <svg width={w} height={h}>
   892→   892→      <polyline points={points} fill="none" stroke={color} strokeWidth="1.5" strokeLinecap="round" />
   893→   893→    </svg>
   894→   894→  )
   895→   895→}
