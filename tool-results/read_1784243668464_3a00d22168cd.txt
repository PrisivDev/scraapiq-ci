   126→function formatCount(n: number): string {
   127→  if (n >= 1_000_000) return `${(n / 1_000_000).toFixed(1)}M`
   128→  if (n >= 1_000) return `${(n / 1_000).toFixed(1)}K`
   129→  return String(n)
   130→}

   142→export function ScraperView() {
   143→  const [engine, setEngine] = useState<Engine>("google-maps")
   144→  const [keyword, setKeyword] = useState("restaurant")
   145→  const [city, setCity] = useState("Abidjan")
   146→  const [commune, setCommune] = useState("Cocody")
   147→  const [neighborhood, setNeighborhood] = useState("")
   148→  const [maxResults, setMaxResults] = useState(10)
   149→  // Facebook-only fields
   150→  const [cookies, setCookies] = useState("")
   151→  const [pageUrl, setPageUrl] = useState("")
   152→
   153→  const [jobs, setJobs] = useState<Record<Engine, JobState | null>>({
   154→    "google-maps": null,
   155→    facebook: null,
   156→  })
   157→  const [pollingEngine, setPollingEngine] = useState<Engine | null>(null)
   158→  const [launching, setLaunching] = useState(false)
   159→  const [selectedPlace, setSelectedPlace] = useState<ScrapedPlace | null>(null)
   160→
   161→  const currentJob = jobs[engine]
   162→  const isRunning = currentJob?.progress.status === "running" || currentJob?.progress.status === "queued"
   163→
   164→  // Polling de l'état du job (par moteur)
   165→  const pollJob = useCallback(async (jobId: string, eng: Engine) => {
   166→    try {
   167→      const res = await fetch(ENDPOINTS[eng].job(jobId), { credentials: "include" })
   168→      if (!res.ok) return
   169→      const data: JobState = await res.json()
   170→      setJobs((prev) => ({ ...prev, [eng]: data }))
   171→
   172→      if (data.progress.status === "running" || data.progress.status === "queued") {
   173→        setTimeout(() => pollJob(jobId, eng), 1500)
   174→      } else {
   175→        setPollingEngine((cur) => (cur === eng ? null : cur))
   176→        if (data.progress.status === "completed" && data.result) {
   177→          toast.success("Scraping terminé", {
   178→            description: `${data.result.stats.uniqueCount} lieu(s) unique(s) extrait(s) en ${(data.result.stats.durationMs / 1000).toFixed(1)}s`,
   179→          })
   180→        } else if (data.progress.status === "failed") {
   181→          toast.error("Scraping échoué", {
   182→            description: data.progress.errors[0] || "Erreur inconnue",
   183→          })
   184→        }
   185→      }
   186→    } catch (err) {
   187→      console.error("[poll] error:", err)
   188→      setPollingEngine((cur) => (cur === eng ? null : cur))
   189→    }
   190→  }, [])
   191→
   192→  // Lance un job
   193→  const launchJob = async () => {
   194→    const isDirectPage = engine === "facebook" && pageUrl.trim().length > 0
   195→    if (!isDirectPage && !keyword.trim()) {
   196→      toast.error(engine === "facebook"
   197→        ? "Veuillez saisir un mot-clé ou une URL de page Facebook"
   198→        : "Veuillez saisir un mot-clé")
   199→      return
   200→    }
   201→
   202→    setLaunching(true)
   203→    try {
   204→      const body: Record<string, unknown> = {
   205→        keyword: keyword.trim() || undefined,
   206→        city: city.trim() || undefined,
   207→        commune: commune.trim() || undefined,
   208→        neighborhood: neighborhood.trim() || undefined,
   209→        maxResults,
   210→      }
   211→      if (engine === "facebook") {
   212→        if (cookies.trim()) body.cookies = cookies.trim()
   213→        if (pageUrl.trim()) body.pageUrl = pageUrl.trim()
   214→      }
   215→
   216→      const res = await fetch(ENDPOINTS[engine].launch, {
   217→        method: "POST",
   218→        headers: { "Content-Type": "application/json" },
   219→        credentials: "include",
   220→        body: JSON.stringify(body),
   221→      })
   222→
   223→      if (!res.ok) {
   224→        const err = await res.json().catch(() => ({}))
   225→        throw new Error(err.error || "Échec du lancement")
   226→      }
   227→
   228→      const data: LaunchResponse = await res.json()
   229→      const { jobId } = data
   230→
   231→      if (engine === "facebook" && data.authenticated === false) {
   232→        toast.warning("Cookies non validés", {
   233→          description: "Le moteur n'a pas pu vérifier les cookies. Le scraping peut échouer ou être limité.",
   234→        })
   235→      } else {
   236→        toast.info("Job de scraping lancé", {
   237→          description: isDirectPage
   238→            ? `Page Facebook : ${pageUrl.trim()}`
   239→            : `Recherche « ${keyword} » à ${commune || city}`,
   240→        })
   241→      }
   242→
   243→      setPollingEngine(engine)
   244→      pollJob(jobId, engine)
   245→    } catch (err) {
   246→      toast.error("Erreur", { description: (err as Error).message })
   247→    } finally {
   248→      setLaunching(false)
   249→    }
   250→  }
   251→
   252→  // Annule le job courant
   253→  const cancelJob = async () => {
   254→    if (!currentJob) return
   255→    try {
   256→      await fetch(ENDPOINTS[engine].job(currentJob.id), {
   257→        method: "DELETE",
   258→        credentials: "include",
   259→      })
   260→      toast.info("Job annulé")
   261→      pollJob(currentJob.id, engine)
   262→    } catch {
   263→      toast.error("Erreur lors de l'annulation")
   264→    }
   265→  }
   266→
   267→  // Récupère les jobs GM récents au montage (FB n'a pas d'endpoint list)
   268→  useEffect(() => {
   269→    fetch("/api/scraper/jobs", { credentials: "include" })
   270→      .then((r) => r.json())
   271→      .then((data) => {
   272→        if (data.jobs && data.jobs.length > 0) {
   273→          pollJob(data.jobs[0].id, "google-maps")
   274→        }
   275→      })
   276→      .catch(() => {})
   277→  }, [pollJob])
   278→
   279→  const hasFbCookies = cookies.trim().length > 0
   280→  const isDirectPageMode = engine === "facebook" && pageUrl.trim().length > 0
   281→
   282→  return (
   283→    <div className="space-y-4">
   284→      {/* Header + engine switch */}
   285→      <div className="flex flex-wrap items-center justify-between gap-3">
   286→        <div>
   287→          <h1 className="text-2xl font-bold tracking-tight flex items-center gap-2">
   288→            <Radar className="h-6 w-6 text-primary" />
   289→            Moteur de scraping
   290→          </h1>
   291→          <p className="text-sm text-muted-foreground mt-0.5">
   292→            {engine === "facebook"
   293→              ? "Facebook · Playwright · Extraction pages & recherche · Déduplication IA"
   294→              : "Google Maps · Playwright · Extraction complète · Déduplication IA · Anti-blocage"}
   295→          </p>
   296→        </div>
   297→        <div className="flex items-center gap-3">
   298→          {isRunning && (
   299→            <Badge variant="outline" className="bg-primary/10 text-primary gap-1.5">
   300→              <Loader2 className="h-3 w-3 animate-spin" />
   301→              Scraping en cours…
   302→            </Badge>
   303→          )}
   304→          <Tabs value={engine} onValueChange={(v) => setEngine(v as Engine)}>
   305→            <TabsList>
   306→              <TabsTrigger value="google-maps" className="gap-1.5">
   307→                <MapPin className="h-3.5 w-3.5" />
   308→                Google Maps
   309→              </TabsTrigger>
   310→              <TabsTrigger value="facebook" className="gap-1.5">
   311→                <FacebookIcon className="h-3.5 w-3.5" />
   312→                Facebook
   313→              </TabsTrigger>
   314→            </TabsList>
   315→          </Tabs>
   316→        </div>
   317→      </div>
   318→
   319→      {/* Formulaire de recherche */}
   320→      <Card>
   321→        <CardHeader className="pb-3">
   322→          <CardTitle className="text-base flex items-center gap-2">
   323→            {engine === "facebook" ? (
   324→              <FacebookIcon className="h-4 w-4 text-orange-600 dark:text-orange-400" />
   325→            ) : (
   326→              <Tag className="h-4 w-4 text-primary" />
   327→            )}
   328→            Critères de recherche{engine === "facebook" ? " — Facebook" : " — Google Maps"}
   329→          </CardTitle>
   330→          <CardDescription className="text-xs">
   331→            {engine === "facebook"
   332→              ? "Le moteur va ouvrir Facebook, charger les pages et extraire les coordonnées (Messenger, WhatsApp, site, etc.)"
   333→              : "Le moteur va ouvrir Google Maps, scroller les résultats et extraire chaque fiche détaillée"}
   334→          </CardDescription>
   335→        </CardHeader>
   336→        <CardContent className="space-y-4">
   337→          {/* Alerte cookies Facebook */}
   338→          {engine === "facebook" && (
   339→            <Alert className="border-orange-300 bg-orange-50 dark:border-orange-900 dark:bg-orange-950/40">
   340→              <AlertTriangle className="h-4 w-4 text-orange-600 dark:text-orange-400" />
   341→              <AlertTitle className="text-orange-800 dark:text-orange-200">
   342→                ⚠️ Connexion Facebook requise
   343→              </AlertTitle>
   344→              <AlertDescription className="text-orange-700 dark:text-orange-300">
   345→                Facebook nécessite des cookies de session (<code className="font-mono">c_user</code>,{" "}
   346→                <code className="font-mono">xs</code>) pour accéder à la plupart des pages.
   347→                Récupérez-les depuis votre navigateur → DevTools → Application → Cookies → facebook.com
   348→              </AlertDescription>
   349→            </Alert>
   350→          )}
   351→
   352→          {/* Bannière warning si FB sans cookies */}
   353→          {engine === "facebook" && !hasFbCookies && (
   354→            <div className="rounded-lg border border-amber-300 bg-amber-50 dark:border-amber-900 dark:bg-amber-950/30 p-3">
   355→              <div className="flex items-start gap-2 text-xs">
   356→                <AlertTriangle className="h-4 w-4 text-amber-600 dark:text-amber-400 mt-0.5 shrink-0" />
   357→                <div>
   358→                  <p className="font-semibold text-amber-800 dark:text-amber-200">
   359→                    Cookies de session non fournis
   360→                  </p>
   361→                  <p className="text-amber-700 dark:text-amber-300 mt-0.5">
   362→                    Sans cookies valides, Facebook limite drastiquement l'accès aux pages et
   363→                    le scraping risque d'échouer ou de renvoyer des résultats partiels.
   364→                  </p>
   365→                </div>
   366→              </div>
   367→            </div>
   368→          )}
   369→
   370→          {/* Champs standards (masqués en mode pageUrl direct) */}
   371→          <div className={cn("grid grid-cols-1 md:grid-cols-12 gap-3", isDirectPageMode && "opacity-50 pointer-events-none")}>
   372→            <div className="md:col-span-4 space-y-1.5">
   373→              <Label className="text-xs flex items-center gap-1.5">
   374→                <Tag className="h-3 w-3" /> Mot-clé {!isDirectPageMode && "*"}
   375→              </Label>
   376→              <Input
   377→                value={keyword}
   378→                onChange={(e) => setKeyword(e.target.value)}
   379→                placeholder="restaurant, pharmacie, banque…"
   380→                disabled={isRunning}
   381→              />
   382→            </div>
   383→            <div className="md:col-span-2 space-y-1.5">
   384→              <Label className="text-xs flex items-center gap-1.5">
   385→                <Building2 className="h-3 w-3" /> Ville
   386→              </Label>
   387→              <Input
   388→                value={city}
   389→                onChange={(e) => setCity(e.target.value)}
   390→                placeholder="Abidjan"
   391→                disabled={isRunning}
   392→              />
   393→            </div>
   394→            <div className="md:col-span-2 space-y-1.5">
   395→              <Label className="text-xs flex items-center gap-1.5">
   396→                <MapPin className="h-3 w-3" /> Commune
   397→              </Label>
   398→              <Input
   399→                value={commune}
   400→                onChange={(e) => setCommune(e.target.value)}
   401→                placeholder="Cocody"
   402→                disabled={isRunning}
   403→              />
   404→            </div>
   405→            <div className="md:col-span-2 space-y-1.5">
   406→              <Label className="text-xs flex items-center gap-1.5">
   407→                <MapPin className="h-3 w-3" /> Quartier
   408→              </Label>
   409→              <Input
   410→                value={neighborhood}
   411→                onChange={(e) => setNeighborhood(e.target.value)}
   412→                placeholder="Riviera 2"
   413→                disabled={isRunning}
   414→              />
   415→            </div>
   416→            <div className="md:col-span-2 space-y-1.5">
   417→              <Label className="text-xs">Max résultats</Label>
   418→              <Select value={String(maxResults)} onValueChange={(v) => setMaxResults(Number(v))} disabled={isRunning}>
   419→                <SelectTrigger><SelectValue /></SelectTrigger>
   420→                <SelectContent>
   421→                  {[5, 10, 20, 50].map((n) => (
   422→                    <SelectItem key={n} value={String(n)}>{n} lieux</SelectItem>
   423→                  ))}
   424→                </SelectContent>
   425→              </Select>
   426→            </div>
   427→          </div>
   428→
   429→          {/* Champs Facebook spécifiques */}
   430→          {engine === "facebook" && (
   431→            <div className="grid grid-cols-1 md:grid-cols-12 gap-3">
   432→              <div className="md:col-span-7 space-y-1.5">
   433→                <Label className="text-xs flex items-center gap-1.5">
   434→                  <FacebookIcon className="h-3 w-3" /> URL de page Facebook (optionnel)
   435→                </Label>
   436→                <Input
   437→                  value={pageUrl}
   438→                  onChange={(e) => setPageUrl(e.target.value)}
   439→                  placeholder="https://www.facebook.com/orangecotedivoire"
   440→                  disabled={isRunning}
   441→                />
   442→                <p className="text-[10px] text-muted-foreground">
   443→                  Si renseignée, la recherche par mot-clé est ignorée (scraping direct de la page).
   444→                </p>
   445→              </div>
   446→              <div className="md:col-span-5 space-y-1.5">
   447→                <Label className="text-xs flex items-center gap-1.5">
   448→                  <AlertCircle className="h-3 w-3" /> Cookies Facebook (optionnel)
   449→                </Label>
   450→                <Textarea
   451→                  value={cookies}
   452→                  onChange={(e) => setCookies(e.target.value)}
   453→                  placeholder="c_user=XXXX; xs=YYYY; datr=ZZZZ; fr=WWW"
   454→                  disabled={isRunning}
   455→                  className="min-h-[42px] text-[11px] font-mono"
   456→                />
   457→              </div>
   458→            </div>
   459→          )}
   460→
   461→          {/* Badges & actions */}
   462→          <div className="flex flex-wrap items-center justify-between gap-3">
   463→            <div className="flex flex-wrap items-center gap-3 text-[11px] text-muted-foreground">
   464→              <span className="flex items-center gap-1">
   465→                <CheckCircle2 className="h-3 w-3 text-primary" />
   466→                {engine === "facebook" ? "Auth par cookies" : "Anti-blocage (CAPTCHA, 429, consent)"}
   467→              </span>
   468→              <span className="flex items-center gap-1">
   469→                <Sparkles className="h-3 w-3 text-primary" /> Dédup IA (Jaro-Winkler + GPS)
   470→              </span>
   471→              <span className="hidden sm:flex items-center gap-1">
   472→                <Activity className="h-3 w-3 text-primary" />
   473→                {engine === "facebook" ? "Extraction Messenger/WhatsApp" : "Backoff exponentiel"}
   474→              </span>
   475→            </div>
   476→            <div className="flex gap-2">
   477→              {isRunning ? (
   478→                <Button variant="destructive" onClick={cancelJob} className="gap-2">
   479→                  <Square className="h-4 w-4" />
   480→                  Annuler
   481→                </Button>
   482→              ) : (
   483→                <Button onClick={launchJob} disabled={launching} className="gap-2">
   484→                  {launching ? <Loader2 className="h-4 w-4 animate-spin" /> : <Play className="h-4 w-4" />}
   485→                  Lancer le scraping
   486→                </Button>
   487→              )}
   488→            </div>
   489→          </div>
   490→        </CardContent>
   491→      </Card>
   492→
   493→      {/* Progression en temps réel */}
   494→      {currentJob && (
   495→        <Card>
   496→          <CardHeader className="pb-3">
   497→            <div className="flex items-center justify-between">
   498→              <CardTitle className="text-base flex items-center gap-2">
   499→                <Activity className="h-4 w-4 text-primary" />
   500→                Progression — Job{" "}
   501→                <span className="font-mono text-xs text-muted-foreground">{currentJob.id}</span>
   502→                <Badge variant="outline" className="text-[10px] gap-1 ml-1">
   503→                  {engine === "facebook" ? (
   504→                    <><FacebookIcon className="h-2.5 w-2.5" /> Facebook</>
   505→                  ) : (
   506→                    <><MapPin className="h-2.5 w-2.5" /> Google Maps</>
   507→                  )}
   508→                </Badge>
   509→              </CardTitle>
   510→              <StatusBadge status={currentJob.progress.status} />
   511→            </div>
   512→          </CardHeader>
   513→          <CardContent className="space-y-4">
   514→            {/* Barre de progression */}
   515→            <div className="space-y-2">
   516→              <div className="flex justify-between text-xs">
   517→                <span className="font-medium capitalize">{currentJob.progress.phase}</span>
   518→                <span className="text-muted-foreground">{currentJob.progress.progress}%</span>
   519→              </div>
   520→              <Progress value={currentJob.progress.progress} className="h-2" />
   521→            </div>
   522→
   523→            {/* Stats live */}

   821→function StatusBadge({ status }: { status: string }) {
   822→  const meta: Record<string, { label: string; icon: React.ElementType; className: string }> = {
   823→    queued: { label: "En file", icon: Clock, className: "bg-muted text-muted-foreground" },
   824→    running: { label: "En cours", icon: Loader2, className: "bg-primary/10 text-primary" },
   825→    completed: { label: "Terminé", icon: CheckCircle2, className: "bg-emerald-100 text-emerald-700" },

   827→    cancelled: { label: "Annulé", icon: Square, className: "bg-muted text-muted-foreground" },

   829→  const m = meta[status] || meta.queued

   833→      <Icon className={cn("h-2.5 w-2.5", status === "running" && "animate-spin")} />
   834→      {m.label}
   835→    </Badge>
   836→  )

  1045→function formatEvent(event: { type: string; [key: string]: unknown }): string {
  1046→  switch (event.type) {
  1047→    // Google Maps events
  1048→    case "start":
  1049→      return "🚀 Démarrage du scraping"
  1050→    case "search-loaded":
  1051→      return `🔍 Page de recherche chargée (${event.resultsOnPage} résultats visibles)`
  1052→    case "scroll":
  1053→      return `📜 Scroll ${event.scrollIndex} → ${event.totalResults} résultats`
  1054→    case "place-extracted":
  1055→      return `✓ Extrait: ${(event.place as { name?: string })?.name || ""} (#${event.index})`
  1056→    case "duplicate-detected":
  1057→      return `↺ Doublon détecté: ${(event.place as { name?: string })?.name || ""} → ${event.duplicateOf}`
  1058→    case "block-detected":
  1059→      return `⚠️ Blocage (${event.reason}) ${event.retrying ? "— retry en cours" : ""}`
  1060→    case "error":
  1061→      return `✗ Erreur: ${event.message}`
  1062→    case "progress":
  1063→      return `📊 Phase: ${event.phase} (${event.progress}%)`
  1064→    case "complete":
  1065→      return `✅ Terminé : ${event.results ? (event.results as unknown[]).length : 0} lieux, ${event.duplicates} doublons, en ${(Number(event.durationMs) / 1000).toFixed(1)}s`
  1066→    case "cancelled":
  1067→      return "⏹️ Annulé"
  1068→    // Facebook events
  1069→    case "fb-login-required":
  1070→      return "🔐 Connexion Facebook requise"
  1071→    case "fb-consent-required":
  1072→      return "🍪 Bannière cookies détectée"
  1073→    case "fb-page-loaded":
  1074→      return `📄 Page chargée: ${event.pageName || ""}`
  1075→    case "fb-search-loaded":
  1076→      return `🔍 Recherche: ${event.resultsCount || 0} pages`
  1077→    case "fb-extracted":
  1078→      return `✓ Extrait: ${(event.place as { name?: string })?.name || event.name || ""}`
  1079→    case "fb-error":
  1080→      return `✗ Erreur: ${event.message}`
  1081→    default:
  1082→      return event.type
  1083→  }
