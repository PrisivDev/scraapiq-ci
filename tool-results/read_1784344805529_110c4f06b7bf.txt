   329→function LevelBadge({ level }: { level: LogEntry["level"] }) {
   330→  const map = {
   331→    info: { label: "INFO", className: "bg-primary/10 text-primary border-primary/20" },

   334→    critical: { label: "CRIT", className: "bg-destructive text-destructive-foreground border-destructive" },
   335→  }
