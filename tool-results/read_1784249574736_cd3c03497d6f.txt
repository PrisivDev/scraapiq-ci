   281→   281→            description: data.progress.errors[0] || "Erreur inconnue",
   282→   282→          })
   283→   283→        }
   284→   284→      }
   285→   285→    } catch (err) {

   287→   287→      setPollingEngine((cur) => (cur === eng ? null : cur))
   288→   288→    }
   289→   289→  }, [])
   290→   290→

   293→   293→    const isDirectPage = engine === "facebook" && pageUrl.trim().length > 0
   294→   294→    const isBusinessDirect =
   295→   295→      engine === "business" && (linkedinSlug.trim().length > 0 || linkedinUrl.trim().length > 0)
   296→   296→
   297→   297→    if (engine === "website") {
   298→   298→      if (!siteUrl.trim()) {

   303→   303→        toast.error("Sélectionnez au moins un type de page à visiter")

   337→   337→          maxPeople,
   338→   338→          extractEmployees,
   339→   339→        }
   340→   340→        description = isBusinessDirect

   345→   345→          keyword: keyword.trim() || undefined,

   347→   347→          commune: commune.trim() || undefined,
   348→   348→          neighborhood: neighborhood.trim() || undefined,
   349→   349→          maxResults,
   350→   350→        }

   357→   357→          : `Recherche « ${keyword} » à ${commune || city}`
   358→   358→      }
   359→   359→
   360→   360→      const res = await fetch(ENDPOINTS[engine].launch, {
   361→   361→        method: "POST",

   367→   367→      if (!res.ok) {

   383→   383→      } else {
   384→   384→        toast.info(engine === "business" ? "Job d'identification lancé" : engine === "website" ? "Robot lancé" : "Job de scraping lancé", {
   385→   385→          description,
   386→   386→        })
   387→   387→      }
   388→   388→
   389→   389→      setPollingEngine(engine)
   390→   390→      pollJob(jobId, engine)

   408→   408→    } catch {
   409→   409→      toast.error("Erreur lors de l'annulation")
   410→   410→    }
   411→   411→  }
   412→   412→
   413→   413→  // Récupère les jobs GM récents au montage (FB n'a pas d'endpoint list)
   414→   414→  useEffect(() => {
   415→   415→    fetch("/api/scraper/jobs", { credentials: "include" })
   416→   416→      .then((r) => r.json())
   417→   417→      .then((data) => {
   418→   418→        if (data.jobs && data.jobs.length > 0) {
   419→   419→          pollJob(data.jobs[0].id, "google-maps")
   420→   420→        }
   421→   421→      })
   422→   422→      .catch(() => {})
   423→   423→  }, [pollJob])
   424→   424→
   425→   425→  const hasFbCookies = cookies.trim().length > 0
   426→   426→  const isDirectPageMode = engine === "facebook" && pageUrl.trim().length > 0
   427→   427→
   428→   428→  return (
   429→   429→    <div className="space-y-4">
   430→   430→      {/* Header + engine switch */}
   431→   431→      <div className="flex flex-wrap items-center justify-between gap-3">
   432→   432→        <div>
   433→   433→          <h1 className="text-2xl font-bold tracking-tight flex items-center gap-2">
   434→   434→            <Radar className="h-6 w-6 text-primary" />
   435→   435→            Moteur de scraping
   436→   436→          </h1>
   437→   437→          <p className="text-sm text-muted-foreground mt-0.5">
   438→   438→            {engine === "facebook"
   439→   439→              ? "Facebook · Playwright · Extraction pages & recherche · Déduplication IA"
