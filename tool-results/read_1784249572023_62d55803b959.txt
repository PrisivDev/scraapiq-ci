   207→function formatCount(n: number): string {
   208→  if (n >= 1_000_000) return `${(n / 1_000_000).toFixed(1)}M`

   278→          })

   281→            description: data.progress.errors[0] || "Erreur inconnue",
   282→          })

   284→      }

   294→    const isBusinessDirect =
   295→      engine === "business" && (linkedinSlug.trim().length > 0 || linkedinUrl.trim().length > 0)
   296→
   297→    if (engine === "website") {
   298→      if (!siteUrl.trim()) {
   299→        toast.error("Veuillez saisir l'URL du site à scraper")
   300→        return
   301→      }
   302→      if (websitePageTypes.length === 0) {
   303→        toast.error("Sélectionnez au moins un type de page à visiter")
   304→        return
   305→      }
   306→    } else if (engine === "business") {
   307→      if (!keyword.trim() && !linkedinSlug.trim() && !linkedinUrl.trim()) {
   308→        toast.error("Veuillez saisir un nom, un slug LinkedIn ou une URL LinkedIn")
   309→        return
   310→      }
   311→    } else if (!isDirectPage && !keyword.trim()) {
   312→      toast.error(engine === "facebook"
   313→        ? "Veuillez saisir un mot-clé ou une URL de page Facebook"
   314→        : "Veuillez saisir un mot-clé")
   315→      return
   316→    }
   317→
   318→    setLaunching(true)
   319→    try {
   320→      let body: Record<string, unknown>
   321→      let description: string
   322→
   323→      if (engine === "website") {
   324→        body = {
   325→          url: siteUrl.trim(),
   326→          pageTypes: websitePageTypes,
   327→          maxPages,
   328→        }

   334→          linkedinSlug: linkedinSlug.trim() || undefined,

   345→          keyword: keyword.trim() || undefined,
   346→          city: city.trim() || undefined,
   347→          commune: commune.trim() || undefined,
   348→          neighborhood: neighborhood.trim() || undefined,
   349→          maxResults,

   359→
   360→      const res = await fetch(ENDPOINTS[engine].launch, {
   361→        method: "POST",
   362→        headers: { "Content-Type": "application/json" },
   363→        credentials: "include",
   364→        body: JSON.stringify(body),
   365→      })
   366→
   367→      if (!res.ok) {
   368→        const err = await res.json().catch(() => ({}))
   369→        throw new Error(err.error || "Échec du lancement")
   370→      }
   371→
   372→      const data: LaunchResponse = await res.json()
   373→      const { jobId } = data
   374→

   376→        toast.warning("Cookies non validés", {
   377→          description: "Le moteur n'a pas pu vérifier les cookies. Le scraping peut échouer ou être limité.",

   379→      } else if (engine === "business" && data.authenticated === false) {
   380→        toast.info("Lancement sans cookies LinkedIn", {
   381→          description: "Fonctionne pour les grandes entreprises publiques (Orange, MTN…). Pour les PME, fournissez des cookies (li_at).",
   382→        })
   383→      } else {
   384→        toast.info(engine === "business" ? "Job d'identification lancé" : engine === "website" ? "Robot lancé" : "Job de scraping lancé", {
   385→          description,
   386→        })
   387→      }
   388→
   389→      setPollingEngine(engine)
   390→      pollJob(jobId, engine)
   391→    } catch (err) {
   392→      toast.error("Erreur", { description: (err as Error).message })
   393→    } finally {
   394→      setLaunching(false)
   395→    }

   477→      {/* Formulaire de recherche */}
   478→      <Card>

   481→            {engine === "facebook" ? (

   484→              <Building2 className="h-4 w-4 text-orange-600 dark:text-orange-400" />

   490→            Critères de recherche{engine === "facebook" ? " — Facebook" : engine === "business" ? " — Business / LinkedIn" : engine === "website" ? " — Site Web / Robot" : " — Google Maps"}

  1388→function StatusBadge({ status }: { status: string }) {

  1395→  }
  1396→  const m = meta[status] || meta.queued

  1400→      <Icon className={cn("h-2.5 w-2.5", status === "running" && "animate-spin")} />
  1401→      {m.label}
  1402→    </Badge>
  1403→  )
